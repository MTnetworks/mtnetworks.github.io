隐藏加密的虚拟硬盘

Window代码

重要提醒：如果要修改盘符D，只需在下面代码中更改前2个D ，千万不要修改第3个D，否则会出现无法隐藏！

如果想做进一步的隐藏，可以把BAT文件转成EXE，或者通过压缩加密BAT文件。解释下下面代码的功能，就是在D盘创建一个虚拟可隐藏的M盘符。

@echo off
set ci=3
echo.
echo 注意： 三次输入错误将退出.
echo.
:1
set /p mima=请输入密码：
if \"%mima%\"==\"password\" goto o
set /a ci-=1
if \"%ci%\"==\"0\" cls&echo.&echo =没密码无法进入=&echo.&pause&echo.&exit
cls&echo.&echo 你还有 %ci% 次机会&echo.&goto 1
:o
cls&echo.
echo= 密码正权确，放行 =
md D:\RECYCLED\UDrives.{25336920-03F9-11CF-8FD0-00AA00686F13}>NUL
if exist M:\NUL goto delete
subst M: D:\RECYCLED\UDrives.{25336920-03F9-11CF-8FD0-00AA00686F13}
start M:\
goto end
:delete
subst /D M:
:end
echo.&pause&exit.

加密版代码：（注意：保存的时候选择ANSI编码，否则会出乱码,密码为password）另存为bat

Linux代码

#!/bin/bash
#==========================================================================
# Linux 文件夹加密保险箱 (UOS / 麒麟 兼容)
# 功能：密码保护 → 挂载隐藏目录为可见目录 → 文件管理器自动打开
#       再次运行 → 卸载并隐藏。完全复刻 Windows bat 的 toggle 逻辑。
#==========================================================================

set -euo pipefail

# ----- 配置 -----
CONFIG_DIR="$HOME/.config/folder-locker"
DATA_ROOT="/media/user/DATA1"                          # 数据盘挂载点（改成你的盘，如 /mnt/DATA）
DATA_DIR="$DATA_ROOT/.private_vault"       # 真实数据存放在此盘（隐藏）
MOUNT_POINT="$HOME/私人保险箱"             # 挂载后可见的目录（家目录下，方便访问）
PASSFILE="$CONFIG_DIR/passhash"
RECOVERY_FILE="$CONFIG_DIR/recovery_hash"
MAX_ATTEMPTS=3

# ----- 工具函数 -----
hash_password() {
    echo -n "$1" | sha256sum | awk '{print $1}'
}

generate_recovery_code() {
    # 生成16位字母数字恢复码，格式: XXXX-XXXX-XXXX-XXXX
    local code=""
    for i in 1 2 3 4; do
        local block=$(head -c 4 /dev/urandom | base64 2>/dev/null | tr -dc 'A-Za-z0-9' | head -c 4)
        [ -n "$block" ] || block=$(tr -dc 'A-Za-z0-9' < /dev/urandom | head -c 4)  # 兼容 busybox
        code="${code}${block}"
        [ $i -lt 4 ] && code="${code}-"
    done
    echo "$code"
}

reset_password_flow() {
    echo ""
    echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
    echo "  密码重置"
    echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
    echo ""
    
    while true; do
        read -s -p "请输入新密码 (6位以上): " pass1
        echo ""
        if [ ${#pass1} -lt 6 ]; then
            echo "⚠️  密码至少需要6位，请重新设置"
            echo ""
            continue
        fi
        read -s -p "请再次输入确认: " pass2
        echo ""
        if [ "$pass1" != "$pass2" ]; then
            echo "⚠️  两次输入不一致，请重新设置"
            echo ""
            continue
        fi
        break
    done
    
    # 更新密码
    hash_password "$pass1" > "$PASSFILE"
    
    # 生成新的恢复码
    local new_code
    new_code=$(generate_recovery_code)
    hash_password "$new_code" > "$RECOVERY_FILE"
    
    echo ""
    echo "✅ 密码重置成功！"
    echo ""
    echo "╔══════════════════════════════════════╗"
    echo "║  🔑 新的恢复密钥（请立即保存！）      ║"
    echo "║                                      ║"
    echo "║      $new_code      ║"
    echo "║                                      ║"
    echo "║  忘记密码时输入此码即可重置。         ║"
    echo "║  丢失无法找回，请妥善保管！           ║"
    echo "╚══════════════════════════════════════╝"
    echo ""
    read -p "按 Enter 继续..."
}

attempt_sudo_mount() {
    # 尝试多种提权方式
    if command -v pkexec &>/dev/null; then
        pkexec mount --bind "$DATA_DIR" "$MOUNT_POINT" 2>/dev/null
    elif command -v gksudo &>/dev/null; then
        gksudo mount --bind "$DATA_DIR" "$MOUNT_POINT" 2>/dev/null
    elif command -v kdesu &>/dev/null; then
        kdesu mount --bind "$DATA_DIR" "$MOUNT_POINT" 2>/dev/null
    else
        sudo mount --bind "$DATA_DIR" "$MOUNT_POINT" 2>/dev/null
    fi
}

attempt_sudo_umount() {
    if command -v pkexec &>/dev/null; then
        pkexec umount "$MOUNT_POINT" 2>/dev/null
    elif command -v gksudo &>/dev/null; then
        gksudo umount "$MOUNT_POINT" 2>/dev/null
    elif command -v kdesu &>/dev/null; then
        kdesu umount "$MOUNT_POINT" 2>/dev/null
    else
        sudo umount "$MOUNT_POINT" 2>/dev/null
    fi
}

open_file_manager() {
    if command -v dde-file-manager &>/dev/null; then
        dde-file-manager "$MOUNT_POINT" &
    elif command -v nautilus &>/dev/null; then
        nautilus "$MOUNT_POINT" &
    elif command -v nemo &>/dev/null; then
        nemo "$MOUNT_POINT" &
    elif command -v thunar &>/dev/null; then
        thunar "$MOUNT_POINT" &
    elif command -v dolphin &>/dev/null; then
        dolphin "$MOUNT_POINT" &
    else
        xdg-open "$MOUNT_POINT" &
    fi
}

# ----- 首次运行：设置密码 -----
if [ ! -f "$PASSFILE" ]; then
    echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
    echo "  首次使用，请设置保险箱密码"
    echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
    echo ""
    
    while true; do
        read -s -p "请输入新密码 (6位以上): " pass1
        echo ""
        if [ ${#pass1} -lt 6 ]; then
            echo "⚠️  密码至少需要6位，请重新设置"
            echo ""
            continue
        fi
        read -s -p "请再次输入确认: " pass2
        echo ""
        if [ "$pass1" != "$pass2" ]; then
            echo "⚠️  两次输入不一致，请重新设置"
            echo ""
            continue
        fi
        break
    done
    
    # 检查数据盘是否存在
    if [ ! -d "$DATA_ROOT" ]; then
        echo ""
        echo "⚠️  数据盘 $DATA_ROOT 不存在！"
        echo "   请先挂载数据盘，或修改脚本第 12 行的 DATA_ROOT。"
        echo ""
        read -p "按 Enter 退出..."
        exit 1
    fi
    
    mkdir -p "$CONFIG_DIR"
    mkdir -p "$DATA_DIR"
    hash_password "$pass1" > "$PASSFILE"
    
    # 生成恢复密钥
    RECOVERY_CODE=$(generate_recovery_code)
    hash_password "$RECOVERY_CODE" > "$RECOVERY_FILE"
    
    chmod 700 "$CONFIG_DIR"
    chmod 600 "$PASSFILE"
    chmod 600 "$RECOVERY_FILE"
    
    echo ""
    echo "✅ 密码设置成功！数据将保存在隐藏目录中。"
    echo ""
    echo "╔══════════════════════════════════════╗"
    echo "║  🔑 恢复密钥（请立即保存！）          ║"
    echo "║                                      ║"
    echo "║      $RECOVERY_CODE      ║"
    echo "║                                      ║"
    echo "║  忘记密码时输入此码即可重置。         ║"
    echo "║  丢失无法找回，请妥善保管！           ║"
    echo "╚══════════════════════════════════════╝"
    echo ""
    read -p "按 Enter 退出..."
    exit 0
fi

# ----- 密码验证 -----
STORED_HASH=$(cat "$PASSFILE")

echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "  保险箱 - 请输入密码解锁"
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo ""
echo "  💡 忘记密码？输入 0 使用恢复密钥"
echo ""

remaining=$MAX_ATTEMPTS
while [ $remaining -gt 0 ]; do
    read -s -p "密码: " input_pass
    echo ""
    
    # 检查是否请求恢复模式
    if [ "$input_pass" = "0" ]; then
        recover_flag=1
        break
    fi
    
    input_hash=$(hash_password "$input_pass")
    
    if [ "$input_hash" = "$STORED_HASH" ]; then
        echo "✅ 密码验证成功！"
        echo ""
        recover_flag=0
        break
    fi
    
    remaining=$((remaining - 1))
    if [ $remaining -eq 0 ]; then
        echo ""
        echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
        echo "  ⛔ 密码错误次数过多"
        echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
        echo ""
        echo "  是否使用恢复密钥重置？ (y/n)"
        read -p "  > " use_recovery
        if [ "$use_recovery" = "y" ] || [ "$use_recovery" = "Y" ]; then
            recover_flag=1
            break
        else
            echo ""
            echo "  已取消。"
            read -p "按 Enter 退出..."
            exit 1
        fi
    fi
    echo "❌ 密码错误，还剩 $remaining 次机会"
    echo ""
done

# ----- 恢复密钥流程 -----
if [ "${recover_flag:-0}" -eq 1 ]; then
    STORED_RECOVERY_HASH=$(cat "$RECOVERY_FILE")
    
    echo ""
    echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
    echo "  恢复密钥验证"
    echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
    echo ""
    echo "  请输入16位恢复密钥（含连字符 -）"
    echo ""
    
    recovery_attempts=3
    while [ $recovery_attempts -gt 0 ]; do
        read -s -p "恢复密钥: " input_recovery
        echo ""
        
        recovery_hash=$(hash_password "$input_recovery")
        
        if [ "$recovery_hash" = "$STORED_RECOVERY_HASH" ]; then
            echo ""
            echo "✅ 恢复密钥验证通过！"
            reset_password_flow
            break
        fi
        
        recovery_attempts=$((recovery_attempts - 1))
        if [ $recovery_attempts -eq 0 ]; then
            echo ""
            echo "⛔ 恢复密钥验证失败，无法访问！"
            echo ""
            read -p "按 Enter 退出..."
            exit 1
        fi
        echo "❌ 恢复密钥错误，还剩 $recovery_attempts 次机会"
        echo ""
    done
    
    # 恢复/重置完成后，重新进入密码验证
    STORED_HASH=$(cat "$PASSFILE")
    
    echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
    echo "  保险箱 - 请输入新密码解锁"
    echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
    echo ""
    
    remaining=$MAX_ATTEMPTS
    while [ $remaining -gt 0 ]; do
        read -s -p "密码: " input_pass
        echo ""
        
        input_hash=$(hash_password "$input_pass")
        
        if [ "$input_hash" = "$STORED_HASH" ]; then
            echo "✅ 密码验证成功！"
            echo ""
            break
        fi
        
        remaining=$((remaining - 1))
        if [ $remaining -eq 0 ]; then
            echo ""
            echo "⛔ 密码错误次数过多，无法访问！"
            echo ""
            read -p "按 Enter 退出..."
            exit 1
        fi
        echo "❌ 密码错误，还剩 $remaining 次机会"
        echo ""
    done
fi

# ----- Toggle: 挂载 或 卸载 -----
if mountpoint -q "$MOUNT_POINT" 2>/dev/null; then
    # 已挂载 → 卸载
    echo "🔒 检测到保险箱已打开，正在锁定..."
    
    # 检查目录是否被占用，若有进程占用则提示
    if lsof "$MOUNT_POINT" 2>/dev/null | grep -q .; then
        echo "⚠️  有程序正在使用保险箱中的文件，请先关闭后再锁定"
        read -p "按 Enter 退出..."
        exit 1
    fi
    
    attempt_sudo_umount
    echo "🔒 保险箱已锁定（数据已隐藏）"
    
else
    # 未挂载 → 挂载
    echo "🔓 正在解锁保险箱..."
    
    if [ ! -d "$DATA_ROOT" ]; then
        echo "⚠️  数据盘 $DATA_ROOT 不存在，请先挂载。"
        echo ""
        read -p "按 Enter 退出..."
        exit 1
    fi
    
    if [ ! -d "$DATA_DIR" ]; then
        mkdir -p "$DATA_DIR"
    fi
    
    if [ ! -d "$MOUNT_POINT" ]; then
        mkdir -p "$MOUNT_POINT"
    fi
    
    if attempt_sudo_mount; then
        echo "🔓 保险箱已解锁 → $MOUNT_POINT"
        echo ""
        open_file_manager
        echo "📁 已打开文件管理器"
        echo ""
        echo "提示：再次运行此脚本即可锁定保险箱"
    else
        echo "⚠️  mount 需要管理员权限，请在弹出的对话框中确认。"
        echo "    如果授权失败，可以手动运行:"
        echo "    sudo mount --bind '$DATA_DIR' '$MOUNT_POINT'"
    fi
fi

echo ""
read -p "按 Enter 退出..."
